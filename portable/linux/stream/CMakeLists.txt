cmake_minimum_required(VERSION 3.25)
project(pgr1_gl_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PGR1_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(SDL3 3.2 CONFIG QUIET)
if(NOT TARGET SDL3::SDL3)
    set(PGR1_BUNDLED_SDL3 "${PGR1_ROOT}/build/linux/_deps/sdl3-src")
    if(NOT EXISTS "${PGR1_BUNDLED_SDL3}/CMakeLists.txt")
        message(FATAL_ERROR "SDL3 unavailable; run scripts/portable-dev.sh linux-tools or install libsdl3-dev")
    endif()
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_UNIX_CONSOLE_BUILD OFF CACHE BOOL "" FORCE)
    set(SDL_X11_XTEST OFF CACHE BOOL "" FORCE)
    set(SDL_X11 ON CACHE BOOL "" FORCE)
    set(SDL_WAYLAND ON CACHE BOOL "" FORCE)
    set(SDL_RENDER ON CACHE BOOL "" FORCE)
    set(SDL_GPU OFF CACHE BOOL "" FORCE)
    set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PGR1_BUNDLED_SDL3}" "${CMAKE_BINARY_DIR}/_sdl3" EXCLUDE_FROM_ALL)
endif()
add_executable(pgr1_stream_viewer viewer.cpp frame_ipc.cpp)
target_link_libraries(pgr1_stream_viewer PRIVATE SDL3::SDL3)
add_executable(pgr1_gl_probe gpu_probe.cpp)
target_link_libraries(pgr1_gl_probe PRIVATE SDL3::SDL3)
include(CTest)
if(BUILD_TESTING)
    add_executable(pgr1_stream_ipc_tests frame_ipc_tests.cpp frame_ipc.cpp)
    add_test(NAME pgr1_stream_ipc COMMAND pgr1_stream_ipc_tests)
endif()
